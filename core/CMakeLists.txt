cmake_minimum_required(VERSION 3.15)
project(morpheus_tracker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(CMAKE_SYSTEM_NAME STREQUAL "Emscripten")
    message(STATUS "Building for Emscripten (WebAssembly)")
    add_subdirectory(wasm)
else()
    message(STATUS "Building native")

    option(BUILD_TESTS "Build native tests" ON)
    if(BUILD_TESTS)
        enable_testing()
        add_subdirectory(../test ${CMAKE_BINARY_DIR}/test)
    endif()

    option(BUILD_BENCHMARK "Build benchmark tools" OFF)
    if(BUILD_BENCHMARK)
        add_subdirectory(../benchmark ${CMAKE_BINARY_DIR}/benchmark)
    endif()
endif()

message(STATUS "")
message(STATUS "==============================================")
message(STATUS "Morpheus Tracker Configuration")
message(STATUS "==============================================")
message(STATUS "System:           ${CMAKE_SYSTEM_NAME}")
message(STATUS "C++ Compiler:     ${CMAKE_CXX_COMPILER}")
message(STATUS "Build type:       ${CMAKE_BUILD_TYPE}")
message(STATUS "==============================================")
message(STATUS "")
